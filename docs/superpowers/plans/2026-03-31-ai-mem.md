# ai-mem Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a persistent, graph-based belief system for Claude Code using Apache AGE (PostgreSQL) as the structural store and Rust for defeasible inference — exposing memory tools to Claude via a stdio MCP server.

**Architecture:** AGE stores the belief graph topology and probability scores (durable authority). ai-mem-core loads relevant subgraphs into Rust, runs defeasible propagation and decay, then writes updated scores back. ai-mem-mcp wraps core as a stdio MCP server that Claude Code calls via tool use.

**Tech Stack:** Rust (sqlx 0.8, tokio 1, uuid 1, chrono 0.4, serde_json 1), PostgreSQL + Apache AGE extension, Agda (spec only, no Haskell runtime needed), JSON-RPC 2.0 over stdio for MCP.

```rust
use anyhow::Result;
use sqlx::{
    postgres::{PgConnectOptions, PgPoolOptions},
    PgPool,
};
use std::str::FromStr;

// public first: unqualified CREATE TABLE → public, not ag_catalog.
// ag_catalog second: AGE operators (agtype operators) resolve correctly.
const SEARCH_PATH: &str = "public,ag_catalog";

fn apply_search_path(opts: PgConnectOptions) -> PgConnectOptions {
    opts.options([("search_path", SEARCH_PATH)])
}

pub async fn connect(dsn: &str) -> Result<PgPool> {
    let opts = apply_search_path(PgConnectOptions::from_str(dsn)?);
    let pool = PgPoolOptions::new()
        .max_connections(10)
        .connect_with(opts)
        .await?;
    Ok(pool)
}
```

```rust
// crates/core/src/graph.rs
use anyhow::{bail, Result};
use chrono::{DateTime, Utc};
use serde::{Deserialize, Serialize};
use uuid::Uuid;

/// Validated probability value in [0, 1].
#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
pub struct Probability(f64);

impl Probability {
    pub fn new(v: f64) -> Result<Self> {
        if !(0.0..=1.0).contains(&v) {
            bail!("probability {} is outside [0, 1]", v);
        }
        Ok(Self(v))
    }

    pub fn value(self) -> f64 {
        self.0
    }

    /// Attenuate by factor, clamping result to [0, 1].
    pub fn attenuate(self, factor: f64) -> Self {
        Self((self.0 * factor).clamp(0.0, 1.0))
    }
}

// ── Node types ────────────────────────────────────────────────────────────────

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Belief {
    pub id: Uuid,
    pub content: String,
    pub probability: Probability,
    pub confidence: Probability,
    pub created_at: DateTime<Utc>,
    pub last_activated_at: DateTime<Utc>,
}

impl Belief {
    pub fn new(content: String, probability: f64, confidence: f64) -> Result<Self> {
        let now = Utc::now();
        Ok(Self {
            id: Uuid::new_v4(),
            content,
            probability: Probability::new(probability)?,
            confidence: Probability::new(confidence)?,
            created_at: now,
            last_activated_at: now,
        })
    }
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Pattern {
    pub id: Uuid,
    pub situation: String,
    pub approach: String,
    pub activation_count: u32,
    pub success_rate: Probability,
    pub created_at: DateTime<Utc>,
}

impl Pattern {
    pub fn new(situation: String, approach: String) -> Result<Self> {
        Ok(Self {
            id: Uuid::new_v4(),
            situation,
            approach,
            activation_count: 0,
            success_rate: Probability::new(0.5)?,
            created_at: Utc::now(),
        })
    }
}

// ── Edge types ────────────────────────────────────────────────────────────────

#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
pub enum EdgeType {
    /// X raises P(Y)
    Supports,
    /// X is a defeater for Y — triggers cascade re-propagation
    Defeats,
    /// Pearl causal edge: X is a structural cause of Y
    Causes,
    /// Symmetric: P(X) and P(Y) cannot both be high without reconciliation
    Contradicts,
}

impl EdgeType {
    pub fn as_str(self) -> &'static str {
        match self {
            Self::Supports => "SUPPORTS",
            Self::Defeats => "DEFEATS",
            Self::Causes => "CAUSES",
            Self::Contradicts => "CONTRADICTS",
        }
    }
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Edge {
    pub from_id: Uuid,
    pub to_id: Uuid,
    pub edge_type: EdgeType,
    pub weight: Probability,
}

impl Edge {
    pub fn new(from_id: Uuid, to_id: Uuid, edge_type: EdgeType, weight: f64) -> Result<Self> {
        Ok(Self {
            from_id,
            to_id,
            edge_type,
            weight: Probability::new(weight)?,
        })
    }
}
```

```rust
// crates/core/src/store.rs
use crate::graph::{Belief, Edge, EdgeType, Pattern, Probability};
use anyhow::{Context, Result};
use sqlx::{PgPool, Row};
use uuid::Uuid;

const GRAPH: &str = "ai_mem";

// ── helpers ───────────────────────────────────────────────────────────────────

/// Execute a Cypher query against the AGE graph. Returns no rows.
async fn cypher_exec(pool: &PgPool, query: &str) -> Result<()> {
    // AGE requires the result set to be declared. Use a dummy column.
    let sql = format!(
        "SELECT * FROM ag_catalog.cypher('{GRAPH}', $q${query}$q$) AS (v ag_catalog.agtype)"
    );
    sqlx::query(&sql).execute(pool).await?;
    Ok(())
}

/// Execute a Cypher query that returns rows as agtype JSON strings.
async fn cypher_rows(pool: &PgPool, query: &str) -> Result<Vec<serde_json::Value>> {
    let sql = format!(
        "SELECT v::text FROM ag_catalog.cypher('{GRAPH}', $q${query}$q$) AS (v ag_catalog.agtype)"
    );
    let rows = sqlx::query(&sql).fetch_all(pool).await?;
    rows.into_iter()
        .map(|r| {
            let s: String = r.try_get(0)?;
            Ok(serde_json::from_str(&s)?)
        })
        .collect()
}

fn uuid_to_str(id: Uuid) -> String {
    id.to_string()
}

// ── beliefs ───────────────────────────────────────────────────────────────────

pub async fn insert_belief(pool: &PgPool, belief: &Belief) -> Result<Uuid> {
    let id = uuid_to_str(belief.id);
    let content = belief.content.replace('\'', "\\'");
    let prob = belief.probability.value();
    let conf = belief.confidence.value();
    let created = belief.created_at.timestamp();
    let activated = belief.last_activated_at.timestamp();

    let q = format!(
        r#"CREATE (b:Belief {{
            id: '{id}',
            content: '{content}',
            probability: {prob},
            confidence: {conf},
            created_at: {created},
            last_activated_at: {activated}
        }}) RETURN b"#
    );
    cypher_exec(pool, &q).await.context("insert_belief")?;
    Ok(belief.id)
}

pub async fn get_belief(pool: &PgPool, id: Uuid) -> Result<Option<Belief>> {
    let id_str = uuid_to_str(id);
    let q = format!(
        "MATCH (b:Belief {{id: '{id_str}'}}) RETURN b"
    );
    let rows = cypher_rows(pool, &q).await?;
    rows.into_iter().next().map(|v| parse_belief(&v)).transpose()
}

pub async fn update_belief_probability(pool: &PgPool, id: Uuid, prob: Probability) -> Result<()> {
    let id_str = uuid_to_str(id);
    let p = prob.value();
    let q = format!(
        "MATCH (b:Belief {{id: '{id_str}'}}) SET b.probability = {p}"
    );
    cypher_exec(pool, &q).await
}

pub async fn update_belief_confidence(pool: &PgPool, id: Uuid, conf: Probability) -> Result<()> {
    let id_str = uuid_to_str(id);
    let c = conf.value();
    let q = format!(
        "MATCH (b:Belief {{id: '{id_str}'}}) SET b.confidence = {c}"
    );
    cypher_exec(pool, &q).await
}

pub async fn list_beliefs(pool: &PgPool) -> Result<Vec<Belief>> {
    let q = "MATCH (b:Belief) RETURN b";
    let rows = cypher_rows(pool, q).await?;
    rows.into_iter().map(|v| parse_belief(&v)).collect()
}

// ── patterns ──────────────────────────────────────────────────────────────────

pub async fn insert_pattern(pool: &PgPool, pattern: &Pattern) -> Result<Uuid> {
    let id = uuid_to_str(pattern.id);
    let situation = pattern.situation.replace('\'', "\\'");
    let approach = pattern.approach.replace('\'', "\\'");
    let sr = pattern.success_rate.value();
    let created = pattern.created_at.timestamp();

    let q = format!(
        r#"CREATE (p:Pattern {{
            id: '{id}',
            situation: '{situation}',
            approach: '{approach}',
            activation_count: 0,
            success_rate: {sr},
            created_at: {created}
        }}) RETURN p"#
    );
    cypher_exec(pool, &q).await.context("insert_pattern")?;
    Ok(pattern.id)
}

// ── edges ─────────────────────────────────────────────────────────────────────

pub async fn insert_edge(pool: &PgPool, edge: &Edge) -> Result<()> {
    let from = uuid_to_str(edge.from_id);
    let to   = uuid_to_str(edge.to_id);
    let w    = edge.weight.value();
    let rel  = edge.edge_type.as_str();

    let q = format!(
        "MATCH (a:Belief {{id: '{from}'}}), (b:Belief {{id: '{to}'}}) \
         CREATE (a)-[:{rel} {{weight: {w}}}]->(b)"
    );
    cypher_exec(pool, &q).await.context("insert_edge")?;
    Ok(())
}

/// Insert a CONTRADICTS edge in both directions (symmetry invariant).
pub async fn insert_contradicts(pool: &PgPool, id_a: Uuid, id_b: Uuid) -> Result<()> {
    let a = uuid_to_str(id_a);
    let b = uuid_to_str(id_b);

    // Forward
    let q1 = format!(
        "MATCH (x:Belief {{id: '{a}'}}), (y:Belief {{id: '{b}'}}) \
         CREATE (x)-[:CONTRADICTS {{weight: 1.0}}]->(y)"
    );
    cypher_exec(pool, &q1).await?;

    // Reverse
    let q2 = format!(
        "MATCH (x:Belief {{id: '{a}'}}), (y:Belief {{id: '{b}'}}) \
         CREATE (y)-[:CONTRADICTS {{weight: 1.0}}]->(x)"
    );
    cypher_exec(pool, &q2).await?;
    Ok(())
}

/// Return all CONTRADICTS pairs as (from_id, to_id).
pub async fn get_contradiction_pairs(pool: &PgPool) -> Result<Vec<(Uuid, Uuid)>> {
    let q = "MATCH (a:Belief)-[:CONTRADICTS]->(b:Belief) RETURN a.id, b.id";
    let sql = format!(
        "SELECT a::text, b::text \
         FROM ag_catalog.cypher('{GRAPH}', $q${q}$q$) AS (a ag_catalog.agtype, b ag_catalog.agtype)"
    );
    let rows = sqlx::query(&sql).fetch_all(pool).await?;
    rows.into_iter()
        .map(|r| {
            let a: String = r.try_get(0)?;
            let b: String = r.try_get(1)?;
            let a = parse_uuid_str(&a)?;
            let b = parse_uuid_str(&b)?;
            Ok((a, b))
        })
        .collect()
}

/// Load all Belief IDs reachable from `start_id` via SUPPORTS or CAUSES edges (BFS).
pub async fn get_downstream_beliefs(
    pool: &PgPool,
    start_id: Uuid,
) -> Result<Vec<(Uuid, f64)>> {
    // Returns (id, incoming_edge_weight) for all downstream nodes
    let id_str = uuid_to_str(start_id);
    let q = format!(
        "MATCH (a:Belief {{id: '{id_str}'}})-[r:SUPPORTS|CAUSES*1..]->(b:Belief) \
         RETURN b.id, r[0].weight"
    );
    let sql = format!(
        "SELECT bid::text, w::text \
         FROM ag_catalog.cypher('{GRAPH}', $q${q}$q$) AS (bid ag_catalog.agtype, w ag_catalog.agtype)"
    );
    let rows = sqlx::query(&sql).fetch_all(pool).await?;
    rows.into_iter()
        .map(|r| {
            let bid: String = r.try_get(0)?;
            let w: String   = r.try_get(1)?;
            let id = parse_uuid_str(&bid)?;
            let weight: f64 = w.trim_matches('"').parse().unwrap_or(1.0);
            Ok((id, weight))
        })
        .collect()
}

/// Return all beliefs with probability and confidence scores (for decay).
pub async fn get_all_beliefs_for_decay(pool: &PgPool) -> Result<Vec<(Uuid, f64, f64, i64)>> {
    let q = "MATCH (b:Belief) RETURN b.id, b.probability, b.confidence, b.last_activated_at";
    let sql = format!(
        "SELECT id::text, prob::text, conf::text, act::text \
         FROM ag_catalog.cypher('{GRAPH}', $q${q}$q$) \
         AS (id ag_catalog.agtype, prob ag_catalog.agtype, conf ag_catalog.agtype, act ag_catalog.agtype)"
    );
    let rows = sqlx::query(&sql).fetch_all(pool).await?;
    rows.into_iter()
        .map(|r| {
            let id:   String = r.try_get(0)?;
            let prob: String = r.try_get(1)?;
            let conf: String = r.try_get(2)?;
            let act:  String = r.try_get(3)?;
            let id   = parse_uuid_str(&id)?;
            let prob: f64 = prob.trim_matches('"').parse().unwrap_or(0.5);
            let conf: f64 = conf.trim_matches('"').parse().unwrap_or(0.5);
            let act: i64  = act.trim_matches('"').parse().unwrap_or(0);
            Ok((id, prob, conf, act))
        })
        .collect()
}

// ── parsers ───────────────────────────────────────────────────────────────────

fn parse_belief(v: &serde_json::Value) -> Result<Belief> {
    let props = v.get("properties").unwrap_or(v);
    let id   = parse_uuid_str(props["id"].as_str().unwrap_or(""))?;
    let prob = props["probability"].as_f64().unwrap_or(0.5);
    let conf = props["confidence"].as_f64().unwrap_or(0.5);
    let content = props["content"].as_str().unwrap_or("").to_string();
    let created_ts = props["created_at"].as_i64().unwrap_or(0);
    let activated_ts = props["last_activated_at"].as_i64().unwrap_or(0);

    use chrono::TimeZone;
    Ok(Belief {
        id,
        content,
        probability: Probability::new(prob)?,
        confidence:  Probability::new(conf)?,
        created_at:       Utc.timestamp_opt(created_ts, 0).single().unwrap_or_else(Utc::now),
        last_activated_at: Utc.timestamp_opt(activated_ts, 0).single().unwrap_or_else(Utc::now),
    })
}

fn parse_uuid_str(s: &str) -> Result<Uuid> {
    let s = s.trim().trim_matches('"');
    Ok(Uuid::parse_str(s)?)
}
```

```rust
// crates/core/src/inference.rs
use crate::{
    graph::Probability,
    store,
};
use anyhow::Result;
use sqlx::PgPool;
use std::collections::HashSet;
use uuid::Uuid;

const CONTRADICTION_EPSILON: f64 = 1e-6;

// ── Pure functions (unit-testable without DB) ─────────────────────────────────

/// P(Y) after defeat by N: P(Y) * (1 - weight * P(N)), clamped to [0,1].
pub fn attenuate_by_defeat(p_y: f64, p_n: f64, weight: f64) -> f64 {
    (p_y * (1.0 - weight * p_n)).clamp(0.0, 1.0)
}

/// Apply exponential confidence decay for `days_elapsed` days.
pub fn apply_decay(confidence: f64, days_elapsed: f64, decay_factor: f64) -> f64 {
    (confidence * decay_factor.powf(days_elapsed)).clamp(0.0, 1.0)
}

/// Returns true if P(X) + P(Y) > 1.0 + ε (active contradiction).
pub fn is_contradicting(p_x: f64, p_y: f64) -> bool {
    p_x + p_y > 1.0 + CONTRADICTION_EPSILON
}

// ── DB-backed operations ──────────────────────────────────────────────────────

/// After inserting a DEFEATS(defeater → defeated, weight) edge, propagate the
/// probability attenuation through all downstream SUPPORTS/CAUSES dependents.
/// Uses BFS with a visited set to avoid revisiting nodes (handles DAGs safely).
pub async fn propagate_defeat(
    pool: &PgPool,
    defeater_id: Uuid,
    defeated_id: Uuid,
    defeat_weight: f64,
) -> Result<()> {
    // Get P(defeater)
    let defeater = store::get_belief(pool, defeater_id).await?;
    let p_n = match defeater {
        Some(b) => b.probability.value(),
        None => return Ok(()), // defeater not found — nothing to propagate
    };

    // Attenuate P(defeated)
    let defeated = store::get_belief(pool, defeated_id).await?;
    let p_y = match defeated {
        Some(b) => b.probability.value(),
        None => return Ok(()),
    };

    let new_p_y = attenuate_by_defeat(p_y, p_n, defeat_weight);
    store::update_belief_probability(pool, defeated_id, Probability::new(new_p_y)?).await?;

    // BFS: propagate through downstream SUPPORTS/CAUSES dependents
    let mut visited: HashSet<Uuid> = HashSet::new();
    visited.insert(defeater_id);
    visited.insert(defeated_id);

    let mut queue = vec![(defeated_id, new_p_y)];

    while let Some((node_id, node_p)) = queue.pop() {
        let downstream = store::get_downstream_beliefs(pool, node_id).await?;
        for (child_id, edge_weight) in downstream {
            if visited.contains(&child_id) {
                continue;
            }
            visited.insert(child_id);

            let child = store::get_belief(pool, child_id).await?;
            if let Some(child_belief) = child {
                // The node's reduced probability propagates as a further attenuation
                let child_p = child_belief.probability.value();
                let new_child_p = attenuate_by_defeat(child_p, node_p, edge_weight);
                store::update_belief_probability(
                    pool,
                    child_id,
                    Probability::new(new_child_p)?,
                )
                .await?;
                queue.push((child_id, new_child_p));
            }
        }
    }

    Ok(())
}

/// Check all CONTRADICTS pairs; return those where both beliefs are in
/// active contradiction (P(X) + P(Y) > 1.0 + ε).
pub async fn detect_active_contradictions(pool: &PgPool) -> Result<Vec<(Uuid, Uuid)>> {
    let pairs = store::get_contradiction_pairs(pool).await?;
    let mut active = Vec::new();

    for (a_id, b_id) in pairs {
        if let (Some(a), Some(b)) = (
            store::get_belief(pool, a_id).await?,
            store::get_belief(pool, b_id).await?,
        ) {
            if is_contradicting(a.probability.value(), b.probability.value()) {
                active.push((a_id, b_id));
            }
        }
    }
    Ok(active)
}

/// Apply confidence decay to all beliefs. For each belief not activated within
/// `threshold_days`, multiply confidence by `decay_factor^(days_since_activation)`.
pub async fn decay_all(pool: &PgPool, decay_factor: f64, threshold_days: f64) -> Result<u32> {
    use chrono::Utc;

    let now = Utc::now().timestamp() as f64;
    let beliefs = store::get_all_beliefs_for_decay(pool).await?;
    let mut decayed = 0u32;

    for (id, _prob, conf, last_activated_ts) in beliefs {
        let days_elapsed = (now - last_activated_ts as f64) / 86400.0;
        if days_elapsed < threshold_days {
            continue;
        }
        let new_conf = apply_decay(conf, days_elapsed, decay_factor);
        store::update_belief_confidence(pool, id, Probability::new(new_conf)?).await?;
        decayed += 1;
    }
    Ok(decayed)
}
```

```agda
-- spec/AiMem.agda
-- Phase 1: Lightweight behavioral spec for ai-mem-core.
-- These are type-checked contracts, not proofs. Invariants are stated
-- as postulates and will be proven as the system matures.

module AiMem where

open import Data.Float  using (Float)
open import Data.String using (String)
open import Data.List   using (List)
open import Data.Product using (_×_)
open import Data.Bool   using (Bool)

-- ── Core types ────────────────────────────────────────────────────────────────

-- A probability value. Invariant (stated, not yet proven):
--   ∀ (p : Probability) → 0.0 ≤ p .value × p .value ≤ 1.0
record Probability : Set where
  field
    value : Float

-- A belief node in the graph.
record Belief : Set where
  field
    id          : String        -- UUID as string
    content     : String
    probability : Probability
    confidence  : Probability

-- A pattern node.
record Pattern : Set where
  field
    id              : String
    situation       : String
    approach        : String
    activationCount : Float     -- u32 approximated as Float for spec
    successRate     : Probability

-- Edge types
data EdgeType : Set where
  Supports    : EdgeType   -- X raises P(Y)
  Defeats     : EdgeType   -- X is a defeater for Y (non-monotonic)
  Causes      : EdgeType   -- Pearl causal edge
  Contradicts : EdgeType   -- symmetric; P(X) + P(Y) cannot both be high

record Edge : Set where
  field
    edgeType : EdgeType
    weight   : Probability

-- A graph is a pair of node lists and edge triples.
record BeliefGraph : Set where
  field
    beliefs  : List Belief
    patterns : List Pattern
    edges    : List (String × Edge × String)   -- (from-id, edge, to-id)

-- ── Operation signatures ──────────────────────────────────────────────────────

postulate
  -- Insert a new Belief into the graph.
  insertBelief : Belief → BeliefGraph → BeliefGraph

  -- Insert a new Pattern.
  insertPattern : Pattern → BeliefGraph → BeliefGraph

  -- Insert a generic edge (SUPPORTS, CAUSES, DEFEATS).
  insertEdge : String → Edge → String → BeliefGraph → BeliefGraph

  -- Insert CONTRADICTS in both directions (symmetry invariant enforced here).
  insertContradicts : String → String → BeliefGraph → BeliefGraph

  -- Defeasible propagation: after a DEFEATS edge from N to Y is inserted,
  -- update P(Y) and all downstream SUPPORTS/CAUSES dependents.
  -- Formula: P(Y)' = P(Y) × (1 - weight × P(N))
  propagateDefeat : String → String → Float → BeliefGraph → BeliefGraph

  -- Return all CONTRADICTS pairs where P(X) + P(Y) > 1.0 + ε.
  detectContradictions : BeliefGraph → List (String × String)

  -- Apply exponential decay to beliefs not activated within threshold days.
  -- Formula: confidence' = confidence × factor^(days_elapsed)
  decayAll : Float → Float → BeliefGraph → BeliefGraph

-- ── Stated invariants (to be proven in future phases) ─────────────────────────

postulate
  -- Probability bounds are preserved through propagation.
  propagation-preserves-bounds :
    ∀ (g : BeliefGraph) (n y : String) (w : Float)
    → let g' = propagateDefeat n y w g
      in Bool   -- placeholder: ∀ belief in g', 0 ≤ probability ≤ 1

  -- CONTRADICTS is symmetric: if (a → b) exists, (b → a) exists.
  contradicts-symmetric :
    ∀ (g : BeliefGraph) (a b : String)
    → let g' = insertContradicts a b g
      in Bool   -- placeholder: both directions present in g'

  -- Defeat propagation terminates (assumes no cycles in SUPPORTS/CAUSES).
  propagation-terminates :
    ∀ (g : BeliefGraph) (n y : String) (w : Float)
    → Bool  -- placeholder: propagateDefeat n y w g is total
```

```rust
// crates/mcp/src/main.rs
use ai_mem_core::AiMemService;
use anyhow::Result;
use serde::{Deserialize, Serialize};
use serde_json::{json, Value};
use std::io::{self, BufRead, Write};
use tokio::sync::OnceCell;
use uuid::Uuid;

static SERVICE: OnceCell<AiMemService> = OnceCell::const_new();

#[derive(Deserialize)]
struct JsonRpcRequest {
    id: Option<Value>,
    method: String,
    params: Option<Value>,
}

#[derive(Serialize)]
struct JsonRpcResponse {
    jsonrpc: &'static str,
    id: Value,
    #[serde(skip_serializing_if = "Option::is_none")]
    result: Option<Value>,
    #[serde(skip_serializing_if = "Option::is_none")]
    error: Option<Value>,
}

impl JsonRpcResponse {
    fn ok(id: Value, result: Value) -> Self {
        Self { jsonrpc: "2.0", id, result: Some(result), error: None }
    }
    fn err(id: Value, code: i32, msg: &str) -> Self {
        Self { jsonrpc: "2.0", id, result: None,
               error: Some(json!({"code": code, "message": msg})) }
    }
}

#[tokio::main]
async fn main() -> Result<()> {
    tracing_subscriber::fmt().with_writer(std::io::stderr).init();

    let dsn = std::env::var("AI_MEM_DSN")
        .unwrap_or_else(|_| "postgresql://ai_mem@localhost/ai_mem".to_string());

    SERVICE.get_or_init(|| async {
        AiMemService::connect(&dsn).await.expect("failed to connect to ai_mem DB")
    }).await;

    let stdin = io::stdin();
    let stdout = io::stdout();
    let mut out = stdout.lock();

    for line in stdin.lock().lines() {
        let line = line?;
        if line.trim().is_empty() { continue; }

        let response = match serde_json::from_str::<JsonRpcRequest>(&line) {
            Ok(req) => handle(req).await,
            Err(e) => JsonRpcResponse::err(Value::Null, -32700, &e.to_string()),
        };

        let resp_str = serde_json::to_string(&response)?;
        writeln!(out, "{}", resp_str)?;
        out.flush()?;
    }
    Ok(())
}

async fn handle(req: JsonRpcRequest) -> JsonRpcResponse {
    let id = req.id.unwrap_or(Value::Null);

async fn dispatch_tool(
    svc: &AiMemService,
    name: &str,
    args: &Value,
    id: Value,
) -> JsonRpcResponse {
    match name {
        "insert_belief" => {
            let content = args["content"].as_str().unwrap_or("").to_string();
            let prob    = args["probability"].as_f64().unwrap_or(0.5);
            let conf    = args["confidence"].as_f64().unwrap_or(0.5);
            match svc.insert_belief(content, prob, conf).await {
                Ok(uuid) => JsonRpcResponse::ok(id, json!({
                    "content": [{"type": "text", "text": uuid.to_string()}]
                })),
                Err(e) => JsonRpcResponse::err(id, -32000, &e.to_string()),
            }
        }

        "insert_pattern" => {
            let situation = args["situation"].as_str().unwrap_or("").to_string();
            let approach  = args["approach"].as_str().unwrap_or("").to_string();
            match svc.insert_pattern(situation, approach).await {
                Ok(uuid) => JsonRpcResponse::ok(id, json!({
                    "content": [{"type": "text", "text": uuid.to_string()}]
                })),
                Err(e) => JsonRpcResponse::err(id, -32000, &e.to_string()),
            }
        }

        "record_support" => {
            let from   = parse_uuid(args, "from_id");
            let to     = parse_uuid(args, "to_id");
            let weight = args["weight"].as_f64().unwrap_or(0.5);
            match (from, to) {
                (Ok(f), Ok(t)) => match svc.record_support(f, t, weight).await {
                    Ok(()) => JsonRpcResponse::ok(id, json!({"content": [{"type": "text", "text": "ok"}]})),
                    Err(e) => JsonRpcResponse::err(id, -32000, &e.to_string()),
                },
                _ => JsonRpcResponse::err(id, -32602, "invalid uuid"),
            }
        }

        "record_defeat" => {
            let defeater = parse_uuid(args, "defeater_id");
            let defeated = parse_uuid(args, "defeated_id");
            let weight   = args["weight"].as_f64().unwrap_or(0.5);
            match (defeater, defeated) {
                (Ok(n), Ok(y)) => match svc.record_defeat(n, y, weight).await {
                    Ok(()) => JsonRpcResponse::ok(id, json!({"content": [{"type": "text", "text": "ok"}]})),
                    Err(e) => JsonRpcResponse::err(id, -32000, &e.to_string()),
                },
                _ => JsonRpcResponse::err(id, -32602, "invalid uuid"),
            }
        }

        "record_contradiction" => {
            let a = parse_uuid(args, "id_a");
            let b = parse_uuid(args, "id_b");
            match (a, b) {
                (Ok(a), Ok(b)) => match svc.record_contradiction(a, b).await {
                    Ok(()) => JsonRpcResponse::ok(id, json!({"content": [{"type": "text", "text": "ok"}]})),
                    Err(e) => JsonRpcResponse::err(id, -32000, &e.to_string()),
                },
                _ => JsonRpcResponse::err(id, -32602, "invalid uuid"),
            }
        }

        "get_contradictions" => {
            match svc.get_contradictions().await {
                Ok(pairs) => {
                    let text = pairs.iter()
                        .map(|(a, b)| format!("{} ↔ {}", a, b))
                        .collect::<Vec<_>>()
                        .join("\n");
                    JsonRpcResponse::ok(id, json!({"content": [{"type": "text", "text": text}]}))
                }
                Err(e) => JsonRpcResponse::err(id, -32000, &e.to_string()),
            }
        }

        "decay_all" => {
            let factor = args["decay_factor"].as_f64().unwrap_or(0.95);
            let days   = args["threshold_days"].as_f64().unwrap_or(7.0);
            match svc.decay_all(factor, days).await {
                Ok(n) => JsonRpcResponse::ok(id, json!({
                    "content": [{"type": "text", "text": format!("decayed {} beliefs", n)}]
                })),
                Err(e) => JsonRpcResponse::err(id, -32000, &e.to_string()),
            }
        }

        "list_beliefs" => {
            match svc.list_beliefs().await {
                Ok(beliefs) => {
                    let text = beliefs.iter()
                        .map(|b| format!("[{}] {} (p={:.2}, c={:.2})",
                            b.id, b.content,
                            b.probability.value(), b.confidence.value()))
                        .collect::<Vec<_>>()
                        .join("\n");
                    JsonRpcResponse::ok(id, json!({"content": [{"type": "text", "text": text}]}))
                }
                Err(e) => JsonRpcResponse::err(id, -32000, &e.to_string()),
            }
        }

        _ => JsonRpcResponse::err(id, -32601, &format!("unknown tool: {}", name)),
    }
}

fn parse_uuid(args: &Value, key: &str) -> Result<Uuid> {
    let s = args[key].as_str().unwrap_or("");
    Ok(Uuid::parse_str(s)?)
}
```

```rust
/// Return up to `limit` beliefs whose content contains any word from `context`,
/// ordered by probability descending.
pub async fn query_relevant(pool: &PgPool, context: &str, limit: usize) -> Result<Vec<Belief>> {
    // Extract individual words for matching
    let words: Vec<&str> = context.split_whitespace().collect();
    if words.is_empty() {
        return Ok(vec![]);
    }

    // Build a Cypher WHERE clause: b.content CONTAINS 'word1' OR b.content CONTAINS 'word2'
    let contains_clauses: Vec<String> = words
        .iter()
        .map(|w| format!("b.content CONTAINS '{}'", w.replace('\'', "\\'")))
        .collect();
    let where_clause = contains_clauses.join(" OR ");

    let q = format!(
        "MATCH (b:Belief) WHERE {where_clause} RETURN b ORDER BY b.probability DESC LIMIT {limit}"
    );
    let rows = cypher_rows(pool, &q).await?;
    rows.into_iter().map(|v| parse_belief(&v)).collect()
}
```
